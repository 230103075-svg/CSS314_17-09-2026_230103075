## Phase 4: Skewed Workload Scheduling (N=8000, p=Pphys=8)

| Strategy | T_wall ms | lambda |
|---|---|---|
| Block static | 6345.6 | 2.342 |
| Cyclic static | 4248.6 | 1.020 |
| Dynamic C=1 | 4048.7 | 1.006 |
| Dynamic C=8 | 4368.6 | 1.010 |
| Dynamic C=64 | 4691.6 | 1.050 |
| Dynamic C=512 | 4993.6 | 1.414 |

### Analysis

- Block static gives the last worker the heaviest items (K grows ~ i^2), so lambda = 2.34 and T_wall is set by the slowest worker. Cyclic static interleaves cheap and heavy items: lambda = 1.02.
- C=1 maximizes balance but every item takes the lock on the shared counter (synchronization + cache-line bouncing of the counter); C=512 amortizes the lock but the final chunk can be a large block of heavy items (tail imbalance, lambda = 1.41).
- Optimal C* = 1 (T_wall = 4048.7 ms, lambda = 1.01): small enough to keep the tail short, large enough that lock cost per item is negligible next to the work per chunk.

```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable


@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float


def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode("utf-8"):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h


def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup):
        fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    samples.sort()
    med = samples[runs // 2]
    return BenchStats(samples[0], med, samples[(3 * runs) // 4] - samples[runs // 4])


def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f"{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}")
    return f"VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})"
```

```python
"""Shared helpers (stdlib only): hardware detection, tables, result saving."""
import glob, json, os, platform, re, subprocess, sys
import multiprocessing as mp
from harness import fnv1a_hash

FAST = bool(os.environ.get("LAB_FAST"))          # smoke-test mode (tiny sizes)
WARMUP = 1 if FAST else int(os.environ.get("LAB_WARMUP", 2))
RUNS = 3 if FAST else int(os.environ.get("LAB_RUNS", 7))
OUT = "out"


def get_id():
    return sys.argv[1] if len(sys.argv) > 1 else os.environ.get("LAB_ID", "0")


def seed_of(sid):
    d = re.sub(r"\D", "", sid)
    return int(d) if d else fnv1a_hash(sid) % 10**9


def _run(cmd):
    try:
        return subprocess.run(cmd, capture_output=True, text=True, timeout=15).stdout
    except Exception:
        return ""


def _sc(name):
    try:
        return int(_run(["sysctl", "-n", name]).strip())
    except Exception:
        return None


def _size(s):
    m = re.match(r"(\d+)\s*([KMG]?)", s.strip().upper())
    if not m:
        return None
    return int(m.group(1)) * {"": 1, "K": 1024, "M": 1024**2, "G": 1024**3}[m.group(2)]


def _ps(expr):
    return _run(["powershell", "-NoProfile", "-Command", expr]).strip()


def detect_pphys():
    """Physical cores. Override with env PPHYS if detection is wrong."""
    if os.environ.get("PPHYS"):
        return int(os.environ["PPHYS"])
    n = 0
    try:
        sysname = platform.system()
        if sysname == "Linux":
            out = _run(["lscpu", "-p=CORE,SOCKET"])
            n = len({l for l in out.splitlines() if l and not l.startswith("#")})
        elif sysname == "Darwin":
            n = _sc("hw.physicalcpu") or 0
        elif sysname == "Windows":
            n = int(_ps("(Get-CimInstance Win32_Processor | Measure-Object -Property NumberOfCores -Sum).Sum"))
    except Exception:
        n = 0
    return n if n > 0 else max(1, mp.cpu_count() // 2)


def cpu_model():
    s = platform.system()
    try:
        if s == "Linux":
            for l in open("/proc/cpuinfo"):
                if l.lower().startswith("model name"):
                    return l.split(":", 1)[1].strip()
        elif s == "Darwin":
            return _run(["sysctl", "-n", "machdep.cpu.brand_string"]).strip()
        elif s == "Windows":
            return _ps("(Get-CimInstance Win32_Processor | Select -First 1).Name")
    except Exception:
        pass
    return platform.processor() or "unknown"


def cache_info():
    """Best-effort bytes for L1d/L2/L3 and line size (None if unknown; env L1D_KB, L2_KB, L3_KB, LINE override)."""
    info = {"l1d": None, "l2": None, "l3": None, "line": 64}
    s = platform.system()
    try:
        if s == "Linux":
            for d in sorted(glob.glob("/sys/devices/system/cpu/cpu0/cache/index*")):
                rd = lambda f: open(os.path.join(d, f)).read().strip()
                lvl, typ, size = rd("level"), rd("type"), _size(rd("size"))
                if lvl == "1" and typ == "Data":
                    info["l1d"] = size
                elif lvl == "2":
                    info["l2"] = size
                elif lvl == "3":
                    info["l3"] = size
                try:
                    info["line"] = int(rd("coherency_line_size"))
                except Exception:
                    pass
        elif s == "Darwin":
            info["l1d"], info["l2"], info["l3"] = _sc("hw.l1dcachesize"), _sc("hw.l2cachesize"), _sc("hw.l3cachesize")
            info["line"] = _sc("hw.cachelinesize") or 64
        elif s == "Windows":
            info["l2"] = int(_ps("(Get-CimInstance Win32_Processor | Select -First 1).L2CacheSize")) * 1024
            info["l3"] = int(_ps("(Get-CimInstance Win32_Processor | Select -First 1).L3CacheSize")) * 1024
    except Exception:
        pass
    for k, e in (("l1d", "L1D_KB"), ("l2", "L2_KB"), ("l3", "L3_KB")):
        if os.environ.get(e):
            info[k] = int(os.environ[e]) * 1024
    if os.environ.get("LINE"):
        info["line"] = int(os.environ["LINE"])
    return info


def fmt_bytes(b):
    if b is None:
        return "n/a (fill manually)"
    return f"{b // 1024} KiB" if b < 1024**2 else f"{b / 1024**2:.1f} MiB"


def p_list(plog):
    ps, p = [], 1
    while p <= plog:
        ps.append(p)
        p *= 2
    if plog not in ps:
        ps.append(plog)
    return ps


def karp_flatt(S, p):
    return (1 / S - 1 / p) / (1 - 1 / p)


def md_table(headers, rows):
    out = ["| " + " | ".join(headers) + " |", "|" + "|".join("---" for _ in headers) + "|"]
    out += ["| " + " | ".join(str(c) for c in r) + " |" for r in rows]
    return "\n".join(out)


def save_phase(n, md, data):
    os.makedirs(OUT, exist_ok=True)
    with open(f"{OUT}/phase{n}.md", "w", encoding="utf-8") as f:
        f.write(md)
    with open(f"{OUT}/phase{n}.json", "w") as f:
        json.dump(data, f, indent=1)
    print(md)
```

```python
from harness import run_benchmark, emit_verify_tag

M = 1_600_000 if L.FAST else 16_000_000      # total read-modify-write steps


def fs_worker(shm_name: str, slot_idx: int, iters: int) -> None:
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    acc = arr[slot_idx]
    for i in range(iters):
        acc = (acc * 6364136223846793005 + (i | 1)) & 0xFFFFFFFFFFFFFFFF
        arr[slot_idx] = acc                    # forces store to shared cache line
    del arr
    shm.close()
```

```python
import math
import multiprocessing as mp
from harness import run_benchmark, emit_verify_tag
import labutil as L

N_STRONG = 60_000 if L.FAST else 1_200_000
N0 = 10_000 if L.FAST else 200_000


def stage_a(n_el):                      # serial: 10 sin/sqrt ops per element
    acc = 0.0
    for i in range(n_el):
        v = 0.1 + i * 1e-6
        for _ in range(5):
            v = math.sqrt(1.0 + math.sin(v))
        acc += v
    return acc


def stage_b(rng):                       # parallel: 30 FMA/trig ops per element, no shared writes
    s, e = rng
    acc = 0.0
    for i in range(s, e):
        v = 0.1 + i * 1e-6
        for _ in range(10):
            v = v * 0.9999991 + 3e-7    # FMA
            w = math.sin(v)             # trig
            v = w * 0.5 + v * 0.5       # FMA
        acc += v
    return acc


def chunks(n, p):
    k = 4 * p
    return [(n * c // k, n * (c + 1) // k) for c in range(k)]


def measure(p, n):
    """Median ms of full pipeline (Stage A serial + Stage B on persistent Pool)."""
    pool = mp.Pool(processes=p)         # created OUTSIDE the timed loop
    try:
        def trial():
            stage_a(int(0.05 * n))
            pool.map(stage_b, chunks(n, p), chunksize=1)
        return run_benchmark(trial, L.WARMUP, L.RUNS).median_ms
    finally:
        pool.close()
        pool.join()


def t_stage_a(n):
    return run_benchmark(lambda: stage_a(int(0.05 * n)), L.WARMUP, L.RUNS).median_ms


if __name__ == "__main__":
    seed = L.seed_of(L.get_id())
    plog, pphys = mp.cpu_count(), L.detect_pphys()
    ps = list(range(1, plog + 1))

    # ---- strong scaling
    ta1 = t_stage_a(N_STRONG)
    T = {p: measure(p, N_STRONG) for p in ps}
    S = {p: T[1] / T[p] for p in ps}
    s_frac = ta1 / T[1]
    cap = 1 / s_frac
    rows = [[p, f"{T[p]:.1f}", f"{S[p]:.2f}", f"{S[p] / p:.2f}"] for p in ps]
    t_strong = L.md_table(["p", "T_p ms", "S_p", "E_p"], rows)

    # ---- weak scaling
    TW, TA = {}, {}
    for p in ps:
        n = p * N0
        TW[p] = measure(p, n)
        TA[p] = t_stage_a(n)
    rows, SG = [], {}
    for p in ps:
        alpha = TA[p] / TW[p]
        SG[p] = p - alpha * (p - 1)
        rows.append([p, p * N0, f"{TW[p]:.1f}", f"{TW[1] / TW[p]:.2f}", f"{alpha:.4f}", f"{SG[p]:.2f}"])
    t_weak = L.md_table(["p", "N(p)", "T_weak ms", "E_weak", "alpha", "S_Gustafson"], rows)

    # ---- inflection Pphys -> Pphys+1
    if pphys < plog and pphys >= 2:
        d_after = S[pphys + 1] - S[pphys]
        d_before = (S[pphys] - S[1]) / (pphys - 1)
        infl = (f"Marginal speedup S({pphys + 1}) - S({pphys}) = {d_after:.2f} versus an average of "
                f"{d_before:.2f} per added process below Pphys ({d_after / d_before * 100:.0f}% of earlier gain): "
                f"beyond {pphys} physical cores extra processes land on SMT siblings (or E-cores) that share "
                f"execution units and L1/L2, so returns diminish.")
    else:
        infl = "Pphys == Plog (or Pphys < 2) on this machine: no SMT boundary to measure."

    tag = emit_verify_tag("PHASE2", seed, S[plog], SG[plog])
    md = (f"## Phase 2: Strong vs Weak Scaling\n\n### Strong (N={N_STRONG})\n\n{t_strong}\n\n"
          f"Stage A(1) = {ta1:.1f} ms, s = {s_frac:.4f}, Amdahl cap S_inf = 1/s = {cap:.1f}; "
          f"observed S_Plog = {S[plog]:.2f} ({S[plog] / cap * 100:.0f}% of the cap). "
          f"At this p the cap is not the limit; hardware (SMT sharing, turbo drop, pool overhead) is.\n\n"
          f"### Weak (N0={N0} per worker)\n\n{t_weak}\n\n### Architectural inflection\n\n{infl}\n\n```\n{tag}\n```\n")
```

```python
import multiprocessing as mp
from multiprocessing import shared_memory
from harness import run_benchmark, emit_verify_tag
import labutil as L

N = 64 if L.FAST else 256
BS_LIST = [8, 16, 32, 64, 96, 128]


def _close(shm):
    try:
        shm.close()
    except BufferError:
        pass


# ---------------- 3A: roofline ceilings
def copy_task(a):
    na, nb, s, e = a
    sa, sb = shared_memory.SharedMemory(name=na), shared_memory.SharedMemory(name=nb)
    mv_a, mv_b = sa.buf, sb.buf
    mv_a[s:e] = mv_b[s:e]
    mv_a = mv_b = None
    _close(sa)
    _close(sb)


def fma_task(_):                                       # 16 independent chains, 32 FLOPs / iteration
    m, c = 1.0000001, 1e-7
    x0 = x1 = x2 = x3 = x4 = x5 = x6 = x7 = 1.0
    x8 = x9 = x10 = x11 = x12 = x13 = x14 = x15 = 1.0
    for _ in range(FLOPS_W // 32):
        x0 = x0 * m + c
        x1 = x1 * m + c
        x2 = x2 * m + c
        x3 = x3 * m + c
        x4 = x4 * m + c
        x5 = x5 * m + c
        x6 = x6 * m + c
        x7 = x7 * m + c
        x8 = x8 * m + c
        x9 = x9 * m + c
        x10 = x10 * m + c
        x11 = x11 * m + c
        x12 = x12 * m + c
        x13 = x13 * m + c
        x14 = x14 * m + c
        x15 = x15 * m + c
    return x0 + x1 + x2 + x3 + x4 + x5 + x6 + x7 + x8 + x9 + x10 + x11 + x12 + x13 + x14 + x15


# ---------------- 3B: GEMM variants on flat double buffers
def gemm_task(a):
    names, variant, r0, r1, BS = a
    shs = [shared_memory.SharedMemory(name=n) for n in names]
    A, B, C = [s.buf[:N * N * 8].cast("d") for s in shs]
    if variant == 1:                                   # naive i-j-k, strided B access
        for i in range(r0, r1):
            ai = i * N
            for j in range(N):
                s = 0.0
                for k in range(N):
                    s += A[ai + k] * B[k * N + j]
                C[ai + j] = s
    elif variant == 2:                                 # i-k-j, hoisted a_ik, unit-stride j
        for i in range(r0, r1):
            ci = i * N
            for j in range(N):
                C[ci + j] = 0.0
            for k in range(N):
                a_ik = A[ci + k]
                bk = k * N
                for j in range(N):
                    C[ci + j] += a_ik * B[bk + j]
    else:                                              # 6-loop blocked, (i,k,j) micro-kernel
        for i in range(r0, r1):
            for j in range(N):
                C[i * N + j] = 0.0
        for ii in range(r0, r1, BS):
            i_max = min(ii + BS, r1)
            for jj in range(0, N, BS):
                j_max = min(jj + BS, N)
                for kk in range(0, N, BS):
                    k_max = min(kk + BS, N)
                    for i in range(ii, i_max):
                        ci = i * N
                        for k in range(kk, k_max):
                            a_ik = A[ci + k]
                            bk = k * N
                            for j in range(jj, j_max):
                                C[ci + j] += a_ik * B[bk + j]
    A.release()
    B.release()
    C.release()
    for s in shs:
        _close(s)
```

```python
import math
import multiprocessing as mp
import time
from harness import run_benchmark, emit_verify_tag
import labutil as L

N = 800 if L.FAST else 8000
W = 1 if L.FAST else int(__import__("os").environ.get("LAB_P4_WARMUP", 1))
R = 3 if L.FAST else int(__import__("os").environ.get("LAB_P4_RUNS", 3))   # heavy workload: fewer runs


def work(i):
    k = 1 + (i * i) // 1600                       # == 1 + floor((i/40)^2)
    x = 0.001 * i + 0.1
    for _ in range(k):
        x = math.sqrt(abs(math.sin(x) * math.cos(x))) + 0.5
    return x


def static_worker(mode, t, p, tarr):
    t0 = time.perf_counter()
    it = range(t * N // p, (t + 1) * N // p) if mode == "block" else range(t, N, p)
    for i in it:
        work(i)
    tarr[t] = time.perf_counter() - t0


def dynamic_worker(counter, C, t, tarr):
    t0 = time.perf_counter()
    while True:
        with counter.get_lock():
            s = counter.value
            counter.value = s + C
        if s >= N:
            break
        for i in range(s, min(s + C, N)):
            work(i)
    tarr[t] = time.perf_counter() - t0


def lam_of(tarr):
    v = list(tarr)
    return max(v) / (sum(v) / len(v))


def run_strategy(kind, p, C=None):
    """Returns (median Twall ms, median lambda)."""
    lams = []

    def trial():
        tarr = mp.Array("d", p)
        if kind == "dynamic":
            counter = mp.Value("i", 0)
            procs = [mp.Process(target=dynamic_worker, args=(counter, C, t, tarr)) for t in range(p)]
        else:
            procs = [mp.Process(target=static_worker, args=(kind, t, p, tarr)) for t in range(p)]
        for pr in procs:
            pr.start()
        for pr in procs:
            pr.join()
        lams.append(lam_of(tarr))

    st = run_benchmark(trial, W, R)
    meas = sorted(lams[W:])
    return st.median_ms, meas[len(meas) // 2]


if __name__ == "__main__":
    seed = L.seed_of(L.get_id())
    pphys = L.detect_pphys()
    res = {}
    res["Block static"] = run_strategy("block", pphys)
    res["Cyclic static"] = run_strategy("cyclic", pphys)
    for C in (1, 8, 64, 512):
        res[f"Dynamic C={C}"] = run_strategy("dynamic", pphys, C)
    dyn = {k: v for k, v in res.items() if k.startswith("Dynamic")}
    best_key = min(dyn, key=lambda k: dyn[k][0])
    Cstar = int(best_key.split("=")[1])
    tag = emit_verify_tag("PHASE4", seed, res["Block static"][0], dyn[best_key][0])
    t = L.md_table(["Strategy", "T_wall ms", "lambda"], [[k, f"{v[0]:.1f}", f"{v[1]:.3f}"] for k, v in res.items()])
    md = (f"## Phase 4: Skewed Workload Scheduling (N={N}, p=Pphys={pphys})\n\n{t}\n\n### Analysis\n\n"
          f"- Block static gives the last worker the heaviest items (K grows ~ i^2), so lambda = {res['Block static'][1]:.2f} "
          f"and T_wall is set by the slowest worker. Cyclic static interleaves cheap and heavy items: lambda = {res['Cyclic static'][1]:.2f}.\n"
          f"- C=1 maximizes balance but every item takes the lock on the shared counter (synchronization + cache-line bouncing of the counter); "
          f"C=512 amortizes the lock but the final chunk can be a large block of heavy items (tail imbalance, lambda = {res['Dynamic C=512'][1]:.2f}).\n"
          f"- Optimal C* = {Cstar} (T_wall = {dyn[best_key][0]:.1f} ms, lambda = {dyn[best_key][1]:.2f}): "
          f"small enough to keep the tail short, large enough that lock cost per item is negligible next to the work per chunk.\n\n```\n{tag}\n```\n")
    L.save_phase(4, md, {"hash": tag, "Cstar": Cstar})
```
